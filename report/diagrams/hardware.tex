% Hardware block diagram of the proposed single-lead ECG device.
\begin{tikzpicture}[
    font=\footnotesize,
    blk/.style={draw, thick, rounded corners=2pt, align=center, fill=white, inner sep=4pt},
    sub/.style={draw, rounded corners=1pt, align=center, fill=gray!8, inner sep=3pt, minimum width=2.5cm},
    arr/.style={-{Stealth[length=2.2mm]}, thick},
    bus/.style={{Stealth[length=2.2mm]}-{Stealth[length=2.2mm]}, thick},
    grp/.style={draw, dashed, rounded corners=3pt, inner sep=6pt},
  ]
  % electrodes + protection
  \node[blk, minimum height=2.6cm, text width=1.7cm] (el) {Електроди\\[3pt]RA, LL\\(вход)\\[3pt]RL\\(RLD)};
  \node[blk, right=0.55cm of el, minimum height=2.6cm, text width=2.0cm] (prot)
    {Входна защита\\[2pt]серийни R,\\ESD диоди\\[2pt]RC филтър\\против алиасинг};

  % ADS1292R
  \node[sub, right=0.9cm of prot, yshift=0.95cm] (pga) {PGA (G = 6)};
  \node[sub, below=0.15cm of pga] (adc) {\(\Delta\Sigma\) АЦП, 24 bit\\500 SPS};
  \node[sub, below=0.15cm of adc] (rld) {RLD усилвател,\\lead-off детекция};
  \node[grp, fit=(pga)(adc)(rld), label={[font=\footnotesize\bfseries]above:ADS1292R}] (afe) {};

  % ESP32-S3
  \node[sub, right=1.7cm of pga, minimum width=3.0cm] (dsp) {Децимация 500\(\rightarrow\)100 Hz\\ВЧ филтър 0.5 Hz (каузален)};
  \node[sub, below=0.15cm of dsp, minimum width=3.0cm] (win) {Буфер 256 отчета\\мащабиране};
  \node[sub, below=0.15cm of win, minimum width=3.0cm] (enc) {CAE енкодер (float32)\\int8 квантизация на \(z\)};
  \node[grp, fit=(dsp)(win)(enc), label={[font=\footnotesize\bfseries]above:ESP32-S3 (PSRAM)}] (mcu) {};

  % PC
  \node[blk, right=2.0cm of mcu, text width=2.4cm, minimum height=2.6cm] (pc)
    {Компютър\\[3pt]деквантиза-\\ция, CAE\\декодер,\\визуализация,\\съхранение};

  % power
  \node[blk, below=1.0cm of afe, text width=6.4cm] (pwr) {Li-Po батерия 3.7 V \(\rightarrow\) нискошумни LDO 3.3 V\\(аналогово и цифрово захранване)};

  % connections
  \draw[arr] (el.east) -- (prot.west);
  \draw[arr] (prot.east) -- ++(0.45,0) |- (pga.west);
  \draw[arr] (rld.west) -- ++(-0.45,0) |- ([yshift=-0.9cm]prot.east) node[pos=0.25, below, font=\scriptsize] {RLD};
  \draw[bus] (afe.east |- adc) -- node[above, font=\scriptsize] {SPI} node[below, font=\scriptsize] {DRDY} (mcu.west |- adc);
  \draw[arr] (mcu.east |- win) -- node[above, font=\scriptsize, align=center] {BLE, Wi-Fi\\или USB} (pc.west |- win);
  \draw[arr] (pwr.north) -- (afe.south);
  \draw[arr] (pwr.east) -| (mcu.south);
\end{tikzpicture}
